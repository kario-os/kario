# /dev/console for it, and /proc, /sys, /dev, /state and /root are its mount points.
# Numeric owners: makedevs looks names up in the image's /etc/passwd, which the
# initramfs does not have (and rootfs.py ignores makedevs failing).

/dev/kmsg	c	600	0	0	1	11	-	-	-

/sys		d	555	0	0	-	-	-	-	-
